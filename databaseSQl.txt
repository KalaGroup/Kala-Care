-- ============================================================================
-- APPROVAL APPLICATION — L1..L5 EMPLOYEE HIERARCHY MIGRATION (SQL Server)
-- ============================================================================
-- The old scheme {Employee, BM, HOD, COO} becomes a 5-stage ladder:
--   L1 = Employee (creator; own records AUTO-APPROVE within their own limit)
--   L2 = first branch approval stage        (per-branch assignees)
--   L3 = second branch approval stage       (per-branch assignees)
--   L4 = HOD  (FIXED for L4; per branch PER CATEGORY: spares / services /
--              spares & services)
--   L5 = COO  (FIXED for L5; the rights-master users — final fallback)
--
-- NOTE: the server applies ALL of this automatically on startup
-- (app/performance_indexes.py -> ensure_schema / ensure_performance_indexes).
-- This script is the same migration for manual execution in SSMS.
-- Every statement is idempotent — safe to run more than once.
-- ============================================================================

-- ----------------------------------------------------------------------------
-- 1) NEW TABLE: approval_stage_approvers  (L2 / L3 assignees per branch)
--    (created automatically by SQLAlchemy create_all; kept here for SSMS use)
-- ----------------------------------------------------------------------------
IF OBJECT_ID('dbo.approval_stage_approvers') IS NULL
BEGIN
    CREATE TABLE dbo.approval_stage_approvers (
        id          INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
        branch      NVARCHAR(20)  NOT NULL,
        stage       NVARCHAR(5)   NOT NULL,          -- 'l2' | 'l3'
        user_id     NVARCHAR(50)  NOT NULL,
        user_name   NVARCHAR(100) NULL,
        updated_by  NVARCHAR(50)  NULL,
        created_at  DATETIME      NULL,
        updated_at  DATETIME      NULL,
        CONSTRAINT uq_apv_stage_br_st_user UNIQUE (branch, stage, user_id)
    );
END;

-- ----------------------------------------------------------------------------
-- 2) approval_applications — rename the per-level audit-trail columns
--    branch_* -> l2_*   |   hod_* -> l4_*   |   coo_* -> l5_*
-- ----------------------------------------------------------------------------
IF COL_LENGTH('dbo.approval_applications', 'l2_action_by') IS NULL AND COL_LENGTH('dbo.approval_applications', 'branch_action_by') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.branch_action_by', 'l2_action_by', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l2_action_by_name') IS NULL AND COL_LENGTH('dbo.approval_applications', 'branch_action_by_name') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.branch_action_by_name', 'l2_action_by_name', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l2_action_at') IS NULL AND COL_LENGTH('dbo.approval_applications', 'branch_action_at') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.branch_action_at', 'l2_action_at', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l2_action_remark') IS NULL AND COL_LENGTH('dbo.approval_applications', 'branch_action_remark') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.branch_action_remark', 'l2_action_remark', 'COLUMN';

IF COL_LENGTH('dbo.approval_applications', 'l4_action_by') IS NULL AND COL_LENGTH('dbo.approval_applications', 'hod_action_by') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.hod_action_by', 'l4_action_by', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l4_action_by_name') IS NULL AND COL_LENGTH('dbo.approval_applications', 'hod_action_by_name') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.hod_action_by_name', 'l4_action_by_name', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l4_action_at') IS NULL AND COL_LENGTH('dbo.approval_applications', 'hod_action_at') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.hod_action_at', 'l4_action_at', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l4_action_remark') IS NULL AND COL_LENGTH('dbo.approval_applications', 'hod_action_remark') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.hod_action_remark', 'l4_action_remark', 'COLUMN';

IF COL_LENGTH('dbo.approval_applications', 'l5_action_by') IS NULL AND COL_LENGTH('dbo.approval_applications', 'coo_action_by') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.coo_action_by', 'l5_action_by', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l5_action_by_name') IS NULL AND COL_LENGTH('dbo.approval_applications', 'coo_action_by_name') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.coo_action_by_name', 'l5_action_by_name', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l5_action_at') IS NULL AND COL_LENGTH('dbo.approval_applications', 'coo_action_at') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.coo_action_at', 'l5_action_at', 'COLUMN';
IF COL_LENGTH('dbo.approval_applications', 'l5_action_remark') IS NULL AND COL_LENGTH('dbo.approval_applications', 'coo_action_remark') IS NOT NULL
    EXEC sp_rename 'dbo.approval_applications.coo_action_remark', 'l5_action_remark', 'COLUMN';

-- ----------------------------------------------------------------------------
-- 3) approval_applications — NEW columns
--    L3 audit trail (brand-new stage), auto-approve flag, and the L4/L5
--    approvers an HO-branch creator chooses on the create form.
-- ----------------------------------------------------------------------------
IF COL_LENGTH('dbo.approval_applications', 'l3_action_by') IS NULL
    ALTER TABLE dbo.approval_applications ADD l3_action_by NVARCHAR(50) NULL;
IF COL_LENGTH('dbo.approval_applications', 'l3_action_by_name') IS NULL
    ALTER TABLE dbo.approval_applications ADD l3_action_by_name NVARCHAR(100) NULL;
IF COL_LENGTH('dbo.approval_applications', 'l3_action_at') IS NULL
    ALTER TABLE dbo.approval_applications ADD l3_action_at DATETIME NULL;
IF COL_LENGTH('dbo.approval_applications', 'l3_action_remark') IS NULL
    ALTER TABLE dbo.approval_applications ADD l3_action_remark NVARCHAR(MAX) NULL;
IF COL_LENGTH('dbo.approval_applications', 'auto_approved') IS NULL
    ALTER TABLE dbo.approval_applications ADD auto_approved BIT NOT NULL CONSTRAINT DF_apv_apps_auto_approved DEFAULT 0;
IF COL_LENGTH('dbo.approval_applications', 'l4_approver_id') IS NULL
    ALTER TABLE dbo.approval_applications ADD l4_approver_id NVARCHAR(50) NULL;
IF COL_LENGTH('dbo.approval_applications', 'l4_approver_name') IS NULL
    ALTER TABLE dbo.approval_applications ADD l4_approver_name NVARCHAR(100) NULL;
IF COL_LENGTH('dbo.approval_applications', 'l5_approver_id') IS NULL
    ALTER TABLE dbo.approval_applications ADD l5_approver_id NVARCHAR(50) NULL;
IF COL_LENGTH('dbo.approval_applications', 'l5_approver_name') IS NULL
    ALTER TABLE dbo.approval_applications ADD l5_approver_name NVARCHAR(100) NULL;

-- ----------------------------------------------------------------------------
-- 4) approval_employee_rules — rename the chain flags and add the L3 flag
--    require_branch -> require_l2   |   require_hod -> require_l4
-- ----------------------------------------------------------------------------
IF COL_LENGTH('dbo.approval_employee_rules', 'require_l2') IS NULL AND COL_LENGTH('dbo.approval_employee_rules', 'require_branch') IS NOT NULL
    EXEC sp_rename 'dbo.approval_employee_rules.require_branch', 'require_l2', 'COLUMN';
IF COL_LENGTH('dbo.approval_employee_rules', 'require_l4') IS NULL AND COL_LENGTH('dbo.approval_employee_rules', 'require_hod') IS NOT NULL
    EXEC sp_rename 'dbo.approval_employee_rules.require_hod', 'require_l4', 'COLUMN';
IF COL_LENGTH('dbo.approval_employee_rules', 'require_l3') IS NULL
    ALTER TABLE dbo.approval_employee_rules ADD require_l3 BIT NOT NULL CONSTRAINT DF_apv_rules_require_l3 DEFAULT 1;

-- ----------------------------------------------------------------------------
-- 5) VALUE MIGRATIONS — legacy names -> the L1..L5 scheme
-- ----------------------------------------------------------------------------
-- authority levels: user -> l1, branch -> l2, hod -> l4, coo -> l5
UPDATE dbo.approval_rights
SET level = CASE level WHEN 'user'   THEN 'l1'
                       WHEN 'branch' THEN 'l2'
                       WHEN 'hod'    THEN 'l4'
                       WHEN 'coo'    THEN 'l5'
                       ELSE level END
WHERE level IN ('user','branch','hod','coo');

-- application statuses: pending_branch -> pending_l2, pending_hod -> pending_l4,
-- pending_coo -> pending_l5 (the new pending_l3 only occurs for new records)
UPDATE dbo.approval_applications
SET status = CASE status WHEN 'pending_branch' THEN 'pending_l2'
                         WHEN 'pending_hod'    THEN 'pending_l4'
                         WHEN 'pending_coo'    THEN 'pending_l5'
                         ELSE status END
WHERE status IN ('pending_branch','pending_hod','pending_coo');

-- creator level snapshot on old records
UPDATE dbo.approval_applications
SET created_by_level = CASE created_by_level WHEN 'user'   THEN 'l1'
                                             WHEN 'branch' THEN 'l2'
                                             WHEN 'hod'    THEN 'l4'
                                             WHEN 'coo'    THEN 'l5'
                                             ELSE created_by_level END
WHERE created_by_level IN ('user','branch','hod','coo');

-- rejection level on old records
UPDATE dbo.approval_applications
SET rejected_at_level = CASE rejected_at_level WHEN 'branch' THEN 'l2'
                                               WHEN 'hod'    THEN 'l4'
                                               WHEN 'coo'    THEN 'l5'
                                               ELSE rejected_at_level END
WHERE rejected_at_level IN ('branch','hod','coo');

-- ----------------------------------------------------------------------------
-- 6) NO AUTO-ASSIGNMENT — every L2 / L3 approver is chosen MANUALLY in the
--    Employee Hierarchy tab. Remove any rows a previous auto-seed created
--    (only seeded rows carry updated_by NULL; rows set from the UI always
--    store the admin who saved them).
-- ----------------------------------------------------------------------------
DELETE FROM dbo.approval_stage_approvers WHERE updated_by IS NULL;

-- ----------------------------------------------------------------------------
-- 6b) LEVEL-WISE AUTHORITY LIMITS + RENAMABLE LEVEL NAMES
--     One row per level (l1..l5) with its display name and limits — the
--     limits apply to EVERY user holding the level (no per-user limits).
--     (Both tables are created automatically by SQLAlchemy create_all;
--     kept here for SSMS use.)
-- ----------------------------------------------------------------------------
IF OBJECT_ID('dbo.approval_level_configs') IS NULL
BEGIN
    CREATE TABLE dbo.approval_level_configs (
        id                   INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
        level                NVARCHAR(5)   NOT NULL,   -- 'l1'..'l5'
        display_name         NVARCHAR(100) NULL,       -- NULL = default name
        max_discount_percent FLOAT         NULL,
        max_credit_days      INT           NULL,
        max_expense_amount   FLOAT         NULL,
        updated_by           NVARCHAR(50)  NULL,
        created_at           DATETIME      NULL,
        updated_at           DATETIME      NULL,
        CONSTRAINT uq_apv_level_cfg UNIQUE (level)
    );
END;

IF OBJECT_ID('dbo.approval_level_expense_limits') IS NULL
BEGIN
    CREATE TABLE dbo.approval_level_expense_limits (
        id              INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
        level           NVARCHAR(5) NOT NULL,          -- 'l1'..'l5'
        expense_type_id INT NOT NULL
            CONSTRAINT FK_apv_lvl_exp_type REFERENCES dbo.approval_expense_types(id) ON DELETE CASCADE,
        max_amount      FLOAT        NULL,
        updated_by      NVARCHAR(50) NULL,
        created_at      DATETIME     NULL,
        updated_at      DATETIME     NULL,
        CONSTRAINT uq_apv_lvl_exp_type UNIQUE (level, expense_type_id)
    );
END;

-- One-time seed: the five level rows with their default names (the COO can
-- rename them from the Authority Limit tab). Limits start empty (= 0).
IF NOT EXISTS (SELECT 1 FROM dbo.approval_level_configs)
BEGIN
    INSERT INTO dbo.approval_level_configs (level, display_name, created_at) VALUES
        ('l1', 'Employee', GETDATE()),
        ('l2', 'Approver', GETDATE()),
        ('l3', 'Approver', GETDATE()),
        ('l4', 'HOD',      GETDATE()),
        ('l5', 'COO',      GETDATE());
END;

-- ----------------------------------------------------------------------------
-- 6b-2) LEVEL LIMITS BECAME PER BRANCH — approval_level_configs.branch:
--       NULL = the global row carrying the renamable level name;
--       set  = that branch row's L1..L4 limits (fresh branch rows start 0).
-- ----------------------------------------------------------------------------
IF COL_LENGTH('dbo.approval_level_configs', 'branch') IS NULL
    ALTER TABLE dbo.approval_level_configs ADD branch NVARCHAR(20) NULL;

IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'UX_apv_lvl_cfg_br'
               AND object_id = OBJECT_ID('dbo.approval_level_configs'))
BEGIN
    DECLARE @lq NVARCHAR(256);
    SELECT TOP 1 @lq = name FROM sys.key_constraints
    WHERE parent_object_id = OBJECT_ID('dbo.approval_level_configs') AND type = 'UQ';
    IF @lq IS NOT NULL EXEC('ALTER TABLE dbo.approval_level_configs DROP CONSTRAINT [' + @lq + ']');
    CREATE UNIQUE NONCLUSTERED INDEX UX_apv_lvl_cfg_br
    ON dbo.approval_level_configs (branch, level);
END;

-- ----------------------------------------------------------------------------
-- 6b-3) CREATOR-ATTACHED CC — extra CC addresses set at submit; auto-added
--       to the result email's CC on approval / rejection.
-- ----------------------------------------------------------------------------
IF COL_LENGTH('dbo.approval_applications', 'cc_emails') IS NULL
    ALTER TABLE dbo.approval_applications ADD cc_emails NVARCHAR(500) NULL;

-- ----------------------------------------------------------------------------
-- 6c) EXPENSE NFAs select MULTIPLE types — expense_type stores the combined
--     breakdown text ("Food: 500; Travel: 1000"); widen the column.
-- ----------------------------------------------------------------------------
IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('dbo.approval_applications')
           AND name = 'expense_type' AND max_length < 800)
    ALTER TABLE dbo.approval_applications ALTER COLUMN expense_type NVARCHAR(400) NULL;

-- ----------------------------------------------------------------------------
-- 7) PERFORMANCE INDEXES for the new stage-approver lookups
--    (also applied automatically by performance_indexes.py)
-- ----------------------------------------------------------------------------
-- list_applications scopes L2/L3 viewers by the branches they approve for
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_apv_stage_user_stage'
               AND object_id = OBJECT_ID('dbo.approval_stage_approvers'))
    CREATE NONCLUSTERED INDEX IX_apv_stage_user_stage
    ON dbo.approval_stage_approvers (user_id, stage) INCLUDE (branch);

-- routing decisions resolve (branch, stage) on every submit / approve
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_apv_stage_branch_stage'
               AND object_id = OBJECT_ID('dbo.approval_stage_approvers'))
    CREATE NONCLUSTERED INDEX IX_apv_stage_branch_stage
    ON dbo.approval_stage_approvers (branch, stage) INCLUDE (user_id, user_name);
