-- 1) Add the new column (only if it doesn't already exist)
IF COL_LENGTH('dbo.asset_detailed', 'emission_norm') IS NULL
    ALTER TABLE dbo.asset_detailed ADD emission_norm NVARCHAR(100) NULL;
GO

-- 2) Backfill emission from the old dynamic extra_data JSON
--    (fills the values you already imported, before the column existed)
UPDATE dbo.asset_detailed
SET emission_norm = LEFT(JSON_VALUE(extra_data, '$."EMISSION NORM"'), 100)
WHERE extra_data IS NOT NULL
  AND ISJSON(extra_data) = 1
  AND JSON_VALUE(extra_data, '$."EMISSION NORM"') IS NOT NULL;
GO
