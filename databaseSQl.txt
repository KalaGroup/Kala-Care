-- FIX (2026-07): drafts have app_no = NULL, but SQL Server unique indexes
-- treat NULL as a value, so a second draft failed with a duplicate-key error.
-- Replace the plain unique index/constraint with a FILTERED unique index that
-- only applies to numbered applications. Applied automatically on startup.

           AND object_id = OBJECT_ID('dbo.approval_applications') AND is_unique = 1)
    DROP INDEX ix_approval_applications_app_no ON dbo.approval_applications;

           AND parent_object_id = OBJECT_ID('dbo.approval_applications'))
    ALTER TABLE dbo.approval_applications DROP CONSTRAINT UQ_approval_applications_app_no;

               AND object_id = OBJECT_ID('dbo.approval_applications'))
